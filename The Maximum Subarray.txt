import java.io.*;
import java.util.*;

public class Solution {

    public static List<Integer> maxSubarray(List<Integer> arr) {
        // Maximum subarray sum
        int currentSum = arr.get(0);
        int maxSubarray = arr.get(0);

        // Maximum subsequence sum
        int maxElement = arr.get(0);
        int maxSubsequence = arr.get(0);

        for (int i = 1; i < arr.size(); i++) {
            int value = arr.get(i);

            // Kadane's algorithm for maximum subarray
            currentSum = Math.max(value, currentSum + value);
            maxSubarray = Math.max(maxSubarray, currentSum);

            // Maximum subsequence
            maxElement = Math.max(maxElement, value);

            if (value > 0) {
                maxSubsequence += value;
            }
        }

        // If all elements are negative, choose the largest element
        if (maxSubsequence <= 0) {
            maxSubsequence = maxElement;
        }

        List<Integer> result = new ArrayList<>();
        result.add(maxSubarray);
        result.add(maxSubsequence);

        return result;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(
                new InputStreamReader(System.in)
        );

        int testCases = Integer.parseInt(br.readLine().trim());

        for (int t = 0; t < testCases; t++) {
            int n = Integer.parseInt(br.readLine().trim());

            String[] input = br.readLine().trim().split(" ");
            List<Integer> arr = new ArrayList<>();

            for (int i = 0; i < n; i++) {
                arr.add(Integer.parseInt(input[i]));
            }

            List<Integer> result = maxSubarray(arr);

            System.out.println(result.get(0) + " " + result.get(1));
        }

        br.close();
    }
}
